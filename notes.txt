HIVE : SQL Analytics on Gold Layer
- Data warehouse software built on top of Hadoop that lets you read, write and manages huge datasets
- Instead of storing data in a traditional database format, Hive puts a structured table layer over raw files (CSV, Parquet, ORC) sitting in distributed environment storage (HDFS, Amazon S3).


Hive alternatives:
- Cloud Data Warehouses & Lakehouses: Snowflake, Google Cloud BigQuery, Databricks
- Open Table Formats: Apache Iceberg and Delta Lake
- Faster SQL & Big Data Engines: Spark, Impala, Presto, ClickHouse
- Enterprise Databases: Oracle Exadata, Teradata, IBM DB2

External Tables
- Allows Hive to query files already stored in HDFS without moving or copying the data into Hive's warehouse directory



CREATE EXTERNAL TABLE IF NOT EXISTS city_sales (
    city STRING,
    total_orders BIGINT,
    total_revenue DOUBLE,
    average_order_value DOUBLE
)
STORED AS PARQUET
LOCATION 'hdfs://namenode:8020/retailx-data-lake/gold/city_sales';


CREATE EXTERNAL TABLE IF NOT EXISTS category_sales (
    category STRING,
    total_orders BIGINT,
    total_revenue DOUBLE,
    average_order_value DOUBLE
)
STORED AS PARQUET
LOCATION 'hdfs://namenode:8020/retailx-data-lake/gold/category_sales';


CREATE EXTERNAL TABLE IF NOT EXISTS customer_sales (
    customer_id STRING,
    name STRING,
    city STRING,
    membership STRING,
    total_orders BIGINT,
    total_spent DOUBLE,
    average_order_value DOUBLE
)
STORED AS PARQUET
LOCATION 'hdfs://namenode:8020/retailx-data-lake/gold/customer_sales';

DESCRIBE city_sales;
DESCRIBE category_sales;
DESCRIBE customer_sales;


SELECT * FROM city_sales LIMIT 5;


SELECT
    city,
    total_orders,
    total_revenue
FROM city_sales
ORDER BY total_revenue DESC
LIMIT 5;

Overall business performance
SELECT
    SUM(total_orders) AS total_orders,
    ROUND(SUM(total_revenue), 2) AS total_revenue,
    ROUND(AVG(average_order_value), 2) AS avg_city_order_value
FROM city_sales;
